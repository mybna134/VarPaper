cmake_minimum_required(VERSION 3.24)
project(varpaper_web LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT CEF_ROOT)
  message(FATAL_ERROR "Set CEF_ROOT to the pinned CEF 135.0.17 Linux x86_64 minimal SDK")
endif()
if(NOT EXISTS "${CEF_ROOT}/include/cef_version.h")
  message(FATAL_ERROR "CEF SDK headers missing")
endif()
file(READ "${CEF_ROOT}/include/cef_version.h" CEF_VERSION_HEADER)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/runtime.json" RUNTIME_MANIFEST)
string(JSON RUNTIME_VERSION GET "${RUNTIME_MANIFEST}" version)
string(FIND "${CEF_VERSION_HEADER}" "\"${RUNTIME_VERSION}\"" VERSION_OFFSET)
if(VERSION_OFFSET EQUAL -1)
  message(FATAL_ERROR "CEF SDK version does not match the optional runtime")
endif()
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" cef_wrapper)
include(FetchContent)
FetchContent_Declare(json URL "${CMAKE_CURRENT_SOURCE_DIR}/../wallpaperengine/vendor/json.tar.gz"
  URL_HASH SHA256=aed69825e143af8a785027ec6e5adfa02c552779c3825ea991e246ab06164dfa)
FetchContent_MakeAvailable(json)
add_executable(varpaper-web-host host.cpp)
target_compile_definitions(varpaper-web-host PRIVATE ${CEF_COMPILER_DEFINES})
target_compile_options(varpaper-web-host PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers -fno-rtti -Werror)
target_include_directories(varpaper-web-host PRIVATE "${CEF_ROOT}")
target_link_libraries(varpaper-web-host PRIVATE libcef_dll_wrapper "${CEF_ROOT}/Release/libcef.so" nlohmann_json::nlohmann_json ${CEF_STANDARD_LIBS})
set_target_properties(varpaper-web-host PROPERTIES SKIP_BUILD_RPATH TRUE INSTALL_RPATH "")
install(TARGETS varpaper-web-host RUNTIME DESTINATION libexec)
